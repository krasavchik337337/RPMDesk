PRAGMA foreign_keys = ON;


-- 1. CUSTOMERS
CREATE TABLE IF NOT EXISTS customers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    first_name TEXT NOT NULL,
    last_name TEXT NOT NULL,
    phone TEXT NOT NULL,
    email TEXT,
    notes TEXT,
    is_active INTEGER NOT NULL DEFAULT 1
        CHECK (is_active IN (0, 1)),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);


-- 2. VEHICLES
CREATE TABLE IF NOT EXISTS vehicles (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    customer_id INTEGER NOT NULL,
    vin TEXT UNIQUE,
    make TEXT NOT NULL,
    model TEXT NOT NULL,
    year INTEGER,
    engine TEXT,
    license_plate TEXT,
    notes TEXT,
    is_active INTEGER NOT NULL DEFAULT 1
        CHECK (is_active IN (0, 1)),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (customer_id)
        REFERENCES customers(id)
        ON DELETE RESTRICT
);


-- 3. SERVICE RECORDS
CREATE TABLE IF NOT EXISTS service_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    customer_id INTEGER NOT NULL,
    vehicle_id INTEGER NOT NULL,
    service_date TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    mileage INTEGER,
    notes TEXT,
    total_cost INTEGER NOT NULL DEFAULT 0
        CHECK (total_cost >= 0),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (customer_id)
        REFERENCES customers(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (vehicle_id)
        REFERENCES vehicles(id)
        ON DELETE RESTRICT
);


-- 4. WORK TYPES
CREATE TABLE IF NOT EXISTS work_types (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,
    description TEXT,
    default_price INTEGER NOT NULL DEFAULT 0
        CHECK (default_price >= 0)

);


-- 5. SERVICE WORKS
CREATE TABLE IF NOT EXISTS service_works (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    service_record_id INTEGER NOT NULL,
    work_type_id INTEGER,
    name TEXT NOT NULL,
    quantity INTEGER NOT NULL DEFAULT 1
        CHECK (quantity > 0),
    unit_price INTEGER NOT NULL
        CHECK (unit_price >= 0),
    total_price INTEGER NOT NULL
        CHECK (total_price >= 0),

    FOREIGN KEY (service_record_id)
        REFERENCES service_records(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (work_type_id)
        REFERENCES work_types(id)
        ON DELETE RESTRICT
);


-- 6. INVENTORY
CREATE TABLE IF NOT EXISTS inventory (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    vehicle_make TEXT,
    brand TEXT,
    category TEXT NOT NULL,
    unit TEXT NOT NULL
        CHECK (unit IN ('piece', 'ml')),
    unit_price INTEGER NOT NULL DEFAULT 0
        CHECK (unit_price >= 0),
    notes TEXT
);


-- 7. STOCK MOVEMENTS
CREATE TABLE IF NOT EXISTS stock_movements (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    inventory_id INTEGER NOT NULL,
    movement_type TEXT NOT NULL
        CHECK (
            movement_type IN (
                'receipt',
                'write_off',
                'correction_in',
                'correction_out'
            )
        ),
    quantity INTEGER NOT NULL
        CHECK (quantity > 0),
    unit_price INTEGER NOT NULL DEFAULT 0
        CHECK (unit_price >= 0),
    service_record_id INTEGER,
    note TEXT,
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (inventory_id)
        REFERENCES inventory(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (service_record_id)
        REFERENCES service_records(id)
        ON DELETE RESTRICT
);


-- 8. USED PARTS
CREATE TABLE IF NOT EXISTS used_parts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    service_record_id INTEGER NOT NULL,
    inventory_id INTEGER NOT NULL,
    stock_movement_id INTEGER NOT NULL UNIQUE,
    quantity INTEGER NOT NULL
        CHECK (quantity > 0),
    unit_price INTEGER NOT NULL
        CHECK (unit_price >= 0),
    total_price INTEGER NOT NULL
        CHECK (total_price >= 0),

    FOREIGN KEY (service_record_id)
        REFERENCES service_records(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (inventory_id)
        REFERENCES inventory(id)
        ON DELETE RESTRICT,

    FOREIGN KEY (stock_movement_id)
        REFERENCES stock_movements(id)
        ON DELETE RESTRICT
);



-- 10. SETTINGS
CREATE TABLE IF NOT EXISTS settings (
    key TEXT PRIMARY KEY,
    value TEXT NOT NULL
);



CREATE INDEX IF NOT EXISTS idx_vehicles_customer
    ON vehicles(customer_id);

CREATE INDEX IF NOT EXISTS idx_service_records_customer
    ON service_records(customer_id);

CREATE INDEX IF NOT EXISTS idx_service_records_vehicle
    ON service_records(vehicle_id);

CREATE INDEX IF NOT EXISTS idx_service_records_date
    ON service_records(service_date);

CREATE INDEX IF NOT EXISTS idx_inventory_vehicle_make
    ON inventory(vehicle_make);

CREATE INDEX IF NOT EXISTS idx_service_works_record
    ON service_works(service_record_id);

CREATE INDEX IF NOT EXISTS idx_stock_movements_inventory
    ON stock_movements(inventory_id);

CREATE INDEX IF NOT EXISTS idx_used_parts_record
    ON used_parts(service_record_id);



-- Prevent stock from becoming negative.
CREATE TRIGGER IF NOT EXISTS prevent_negative_stock
BEFORE INSERT ON stock_movements
WHEN NEW.movement_type IN ('write_off', 'correction_out')
BEGIN
    SELECT CASE
        WHEN (
            COALESCE((
                SELECT SUM(
                    CASE
                        WHEN movement_type IN ('receipt', 'correction_in')
                            THEN quantity
                        ELSE -quantity
                    END
                )
                FROM stock_movements
                WHERE inventory_id = NEW.inventory_id
            ), 0)
        ) < NEW.quantity
        THEN RAISE(ABORT, 'Insufficient stock')
    END;
END;


-- Stock movements cannot be edited.
CREATE TRIGGER IF NOT EXISTS prevent_stock_movement_update
BEFORE UPDATE ON stock_movements
BEGIN
    SELECT RAISE(ABORT, 'Stock movements cannot be updated');
END;


-- Stock movements cannot be deleted.
CREATE TRIGGER IF NOT EXISTS prevent_stock_movement_delete
BEFORE DELETE ON stock_movements
BEGIN
    SELECT RAISE(ABORT, 'Stock movements cannot be deleted');
END;
